## 2. 顺序表（动态数组）

### 问题引出

静态数组存在空间不足或浪费的问题。顺序表（动态数组）可以动态调整空间大小。

### 设计思路

```
1. 初始分配较小空间
2. 数据满时重新申请更大的空间（通常 ×1.5）
3. 将旧数据拷贝到新空间
4. 释放旧空间
```

### 整型动态数组

```c
#include <stdio.h>
#include <stdlib.h>

int *insert(int *count, int *arr, int num)
{
    // 申请新空间（比原来多1个）
    int *new = (int *)malloc(sizeof(int) * (*count + 1));
    if (new == NULL)
        return NULL;

    // 拷贝旧数据
    memcpy(new, arr, sizeof(int) * (*count));

    // 新数据放到末尾
    new[*count] = num;
    (*count)++;

    // 释放旧空间
    free(arr);
    return new;
}

int main(void)
{
    int count = 0;
    int *arr = NULL;

    // 循环添加数据
    for (int i = 0; i < 10; i++)
    {
        int num = rand() % 100;
        arr = insert(&count, arr, num);
    }

    // 遍历
    for (int i = 0; i < count; i++)
        printf("%d ", arr[i]);
    printf("\n");

    free(arr);
    return 0;
}
```

### 结构体动态数组

```c
// 学生结构体
struct cls_t {
    char name[64];
    int id;
    int age;
    char sex;
    float linux_score;
    float c_score;
    float sum;
};

// 通用插入函数
struct cls_t *insert(int *count, struct cls_t *prev, struct cls_t cls)
{
    struct cls_t *new = NULL;

    // 新申请内存空间（原来数据 + 1个新数据）
    new = (struct cls_t *)malloc(sizeof(struct cls_t) * (*count + 1));
    if (new == NULL)
        return NULL;

    // 原来的数据复制到新空间
    memcpy(new, prev, sizeof(struct cls_t) * (*count));

    // 释放原来数据空间
    free(prev);

    // 接受新数据保存到空间的结尾
    new[*count] = cls;

    (*count)++;
    return new;
}
```

### 泛型实现（void*封装）

使用 `void*` 可以实现通用的动态数组，适配任意数据类型：

```c
// 通用插入函数 — 适用于任意数据类型
void *insert(int *count, void *prev, void *data, int size)
{
    void *new = NULL;

    // 申请新空间
    new = malloc(size * (*count + 1));
    if (new == NULL)
        return NULL;

    // 拷贝旧数据
    memcpy(new, prev, size * (*count));

    // 追加新数据（注意偏移量计算）
    memcpy(new + *count * size, data, size);

    (*count)++;

    free(prev);
    return new;
}
```

**使用示例：**

```c
// === 使用int类型 ===
int count = 0;
int *arr = NULL;
int num = 123;
arr = insert(&count, arr, &num, sizeof(int));

// === 使用结构体类型 ===
struct cls_t cls;
int cls_count = 0;
struct cls_t *students = NULL;
students = insert(&cls_count, students, &cls, sizeof(struct cls_t));
```

**关键参数说明：**

| 参数 | 含义 |
|------|------|
| `count` | 数据个数的指针，用于跟踪数组大小 |
| `prev` | 原来的数据空间地址 |
| `data` | 新添加的数据地址 |
| `size` | 单个数据元素的大小（sizeof） |

**内存偏移计算：**
```c
memcpy(new + *count * size, data, size);
// new 是 void*，偏移 *count * size 个字节
// 然后拷贝 size 个字节
```

## 3. 泛型顺序表框架设计

### 3.1 结构体定义与回调函数类型

```c
// 遍历回调：对每个元素执行操作
typedef void (darr_op_t)(const void *);

// 比较回调：比较两个元素，返回0相等、>0大于、<0小于
// 第一个参数：内部数据，第二个参数：外部key
typedef int (darr_cmp_t)(const void *, const void *);

// 修改回调：用new数据修改old指向的元素
typedef void (darr_mod_t)(void *, const void *);

// 排序回调：比较两个内部元素的顺序
typedef int (darr_sort_t)(const void *, const void *);

// 顺序表结构体
typedef struct darr_t {
    void *data;     // 数据存储空间
    int size;       // 单个数据的类型大小
    int num;        // 当前数据个数
} DARR;
```

### 3.2 创建与销毁

```c
DARR *darr_creat(int size)
{
    DARR *handle = (DARR *)malloc(sizeof(DARR));
    // 容错处理
    handle->data = NULL;
    handle->size = size;
    handle->num = 0;
    return handle;
}

void darr_destroy(DARR **handle)
{
    free((*handle)->data);
    (*handle)->data = NULL;
    free(*handle);
    *handle = NULL;
}
```

### 3.3 插入操作

插入的核心思路：申请 `size * (num + 1)` 的新空间 → 拷贝旧数据并留出空位 → 填入新数据 → 释放旧空间。

```c
// 尾插：新数据追加到末尾
void darr_insert_end(DARR *handle, const void *data)
{
    void *new = malloc(handle->size * (handle->num + 1));
    // 拷贝旧数据到前面
    memcpy(new, handle->data, handle->size * handle->num);
    // 拷贝新数据到末尾
    memcpy(new + handle->num * handle->size, data, handle->size);
    free(handle->data);
    handle->data = new;
    handle->num++;
}

// 头插：新数据放在开头
void darr_insert_front(DARR *handle, const void *data)
{
    void *new = malloc(handle->size * (handle->num + 1));
    // 旧数据整体后移 size 字节
    memcpy(new + handle->size, handle->data, handle->size * handle->num);
    // 新数据放在开头
    memcpy(new, data, handle->size);
    free(handle->data);
    handle->data = new;
    handle->num++;
}

// 指定位置插入：新数据插入到 index 位置
void darr_insert_index(DARR *handle, const void *data, int index)
{
    void *new = malloc(handle->size * (handle->num + 1));
    // 1> 拷贝前半部分 [0, index)
    memcpy(new, handle->data, handle->size * index);
    // 2> 拷贝新数据到 index 位置
    memcpy(new + handle->size * index, data, handle->size);
    // 3> 拷贝后半部分 [index, num)
    memcpy(new + handle->size * (index + 1),
           handle->data + handle->size * index,
           handle->size * (handle->num - index));
    free(handle->data);
    handle->data = new;
    handle->num++;
}
```

**三段式拷贝图示（index=2, num=5）：**
```
旧数据: [0][1][2][3][4]
新数据: [0][1][X][2][3][4]
         前半  新  后半
```

### 3.4 删除操作

按照比较回调函数找到目标元素，申请 `size * (num - 1)` 空间，跳过目标元素拷贝其余数据。

```c
// 从前往后删除第一个匹配项
void darr_del_front(DARR *handle, const void *key, darr_cmp_t *cmp)
{
    for (int i = 0; i < handle->num; i++)
    {
        if (cmp(handle->data + i * handle->size, key) == 0)
        {
            void *new = malloc(handle->size * (handle->num - 1));
            memcpy(new, handle->data, handle->size * i);
            memcpy(new + handle->size * i,
                   handle->data + handle->size * (i + 1),
                   handle->size * (handle->num - i - 1));
            free(handle->data);
            handle->data = new;
            handle->num--;
            return;
        }
    }
}

// 从后往前删除第一个匹配项
void darr_del_end(DARR *handle, const void *key, darr_cmp_t *cmp)
{
    for (int i = handle->num - 1; i >= 0; i--)
    {
        if (cmp(handle->data + i * handle->size, key) == 0)
        {
            // 同上拷贝逻辑
            // ...
        }
    }
}

// 按索引删除
void darr_del_index(DARR *handle, int index);

// 删除所有匹配项（关键：删除后 i-- 回退，继续检查当前位置）
void darr_del_all(DARR *handle, const void *key, darr_cmp_t *cmp)
{
    for (int i = 0; i < handle->num; i++)
    {
        if (cmp(handle->data + i * handle->size, key) == 0)
        {
            // ... 删除逻辑 ...
            handle->num--;
            i--;  // 回退，因为后面的元素前移了一位
        }
    }
}
```

### 3.5 查找操作

```c
// 从前往后找第一个匹配项，返回元素指针
void *darr_find_front(DARR *handle, const void *key, darr_cmp_t *cmp)
{
    for (int i = 0; i < handle->num; i++)
    {
        if (cmp(handle->data + i * handle->size, key) == 0)
            return handle->data + i * handle->size;
    }
    return NULL;
}

// 从后往前找第一个匹配项
void *darr_find_end(DARR *handle, const void *key, darr_cmp_t *cmp);

// 按索引查找
void *darr_find_index(DARR *handle, int index);

// 查找所有匹配项，返回新的 DARR 句柄
DARR *darr_find_all(DARR *handle, const void *key, darr_cmp_t *cmp)
{
    DARR *ind = darr_creat(handle->size);
    for (int i = 0; i < handle->num; i++)
    {
        if (cmp(handle->data + i * handle->size, key) == 0)
            darr_insert_front(ind, handle->data + i * handle->size);
    }
    return ind;
}
```

### 3.6 修改操作

```c
// 从前往后修改第一个匹配项
void darr_mod_front(DARR *handle, const void *old_key, darr_cmp_t *cmp,
                    void *new_key, darr_mod_t *mod)
{
    for (int i = 0; i < handle->num; i++)
    {
        if (cmp(handle->data + i * handle->size, old_key) == 0)
        {
            mod(handle->data + i * handle->size, new_key);
            return;
        }
    }
}

// 按索引修改
void darr_mod_index(DARR *handle, int index, void *new_key, darr_mod_t *mod);

// 修改所有匹配项

```

### 3.7 排序操作（冒泡排序）

```c
// 升序排列（冒泡排序，memmove 交换元素）
void darr_sort_ascending(DARR *handle, darr_sort_t *sort)
{
    void *tmp = malloc(handle->size);

    for (int i = 0; i < handle->num; i++)
    {
        for (int j = i + 1; j < handle->num; j++)
        {
            // 通过回调函数比较两个元素
            if (sort(handle->data + i * handle->size,
                     handle->data + j * handle->size) > 0)
            {
                // 交换 data[i] 和 data[j]
                memmove(tmp, handle->data + i * handle->size, handle->size);
                memmove(handle->data + i * handle->size,
                        handle->data + j * handle->size, handle->size);
                memmove(handle->data + j * handle->size, tmp, handle->size);
            }
        }
    }
    free(tmp);
}

// 降序排列（接口预留）
void darr_sort_descending(DARR *handle, darr_sort_t *sort);
```

### 3.8 文件存储与加载（二进制）

存储和加载使用统一的二进制格式：`[size][num][data...]`

```c
// 存储到文件
void darr_store(DARR *handle, const char *path)
{
    FILE *fp = fopen(path, "w");

    fwrite(&handle->size, sizeof(handle->size), 1, fp);   // 数据类型大小
    fwrite(&handle->num, sizeof(handle->num), 1, fp);      // 数据个数
    fwrite(handle->data, handle->size, handle->num, fp);   // 数据内容

    fclose(fp);
}

// 从文件加载
DARR *darr_load(const char *path)
{
    FILE *fp = fopen(path, "r");
    int size, num;

    fread(&size, sizeof(size), 1, fp);
    fread(&num, sizeof(num), 1, fp);

    DARR *handle = darr_creat(size);
    handle->data = malloc(size * num);
    fread(handle->data, size, num, fp);
    handle->num = num;

    fclose(fp);
    return handle;
}
```

**二进制文件格式：**
```
┌──────────┬──────────┬─────────────────┐
│  size    │   num    │  data (size×num)│
│ (4字节)   │ (4字节)   │  (变长)          │
└──────────┴──────────┴─────────────────┘
```

### 3.9 完整演示：学生管理系统

```c
// 学生结构体
struct cls_t {
    char name[64];
    int id;
    int age;
    char sex;
    float lin;     // Linux成绩
    float c;       // C成绩
    float sum;     // 总分
};

// ===== 比较函数（cmp）=====
int cmp_name(const void *d1, const void *d2) {
    return strcmp(((struct cls_t *)d1)->name, (char *)d2);
}
int cmp_id(const void *d1, const void *d2) {
    return ((struct cls_t *)d1)->id - *(int *)d2;
}
int cmp_age(const void *d1, const void *d2) {
    return ((struct cls_t *)d1)->age - *(int *)d2;
}
int cmp_sum(const void *d1, const void *d2) {
    float f = ((struct cls_t *)d1)->sum - *(float *)d2;
    if (f > -0.005 && f < 0.005) return 0;  // float容差比较
    return (f > 0.00001) ? 1 : -1;
}

// ===== 修改函数（mod）=====
void mod_age(void *d1, const void *d2) {
    ((struct cls_t *)d1)->age = *(int *)d2;
}
void mod_name(void *d1, const void *d2) {
    strcpy(((struct cls_t *)d1)->name, (char *)d2);
}

// ===== 排序函数（sort）=====
int sort_sum(const void *d1, const void *d2) {
    float f = ((struct cls_t *)d1)->sum - ((struct cls_t *)d2)->sum;
    if (f > -0.005 && f < 0.005) return 0;
    return (f > 0.00001) ? 1 : -1;
}

// ===== 遍历打印 =====
void ls(const void *data) {
    struct cls_t *c = (struct cls_t *)data;
    printf("name:%s id:%d age:%d sex:%c lin:%.2f c:%.2f sum:%.2f\n",
           c->name, c->id, c->age, c->sex, c->lin, c->c, c->sum);
}

int main(void)
{
    struct cls_t cls;
    DARR *handle = darr_creat(sizeof(struct cls_t));

    // 随机生成10个学生并插入
    for (int i = 0; i < 10; i++) {
        snprintf(cls.name, sizeof(cls.name), "cls_%c%c",
                 rand() % 26 + 'A', rand() % 26 + 'a');
        cls.id = 1122000 + rand() % 1000;
        cls.age = rand() % 3 + 17;
        cls.sex = "MF"[rand() % 2];
        cls.lin = 30.0 * (rand() / (RAND_MAX + 1.0)) + 70;
        cls.c = 20.0 * (rand() / (RAND_MAX + 1.0)) + 80;
        cls.sum = cls.lin + cls.c;
        darr_insert_front(handle, &cls);
    }

    // 按总分排序并打印
    darr_sort_ascending(handle, sort_sum);
    darr_travel(handle, ls);

    // 存储到文件
    darr_store(handle, "./stu.db");

    // 按姓名查找
    void *p = darr_find_front(handle, "tom", cmp_name);

    // 按id修改姓名
    int id = 1122123;
    darr_mod_front(handle, &id, cmp_id, "newname", mod_name);

    darr_destroy(&handle);
    return 0;
}
```

### 3.10 浮点数比较注意事项

浮点数不能用 `==` 直接比较，需要容差范围：

```c
// 错误写法
if (a == b)  // 浮点数精度问题，可能永远不相等

// 正确写法：容差 ±0.005
float f = a - b;
if (f > -0.005 && f < 0.005)
    return 0;   // 视为相等
else if (f > 0.00001)
    return 1;   // a > b
else
    return -1;  // a < b
```

### 3.11 API 汇总

| 分类 | 函数 | 说明 |
|------|------|------|
| 创建 | `darr_creat(size)` | 创建指定类型大小的顺序表 |
| 插入 | `darr_insert_end/front/index` | 尾插/头插/指定位置插入 |
| 删除 | `darr_del_front/end/index/all` | 按cmp/key删除 |
| 查找 | `darr_find_front/end/index/all` | 按cmp/key查找 |
| 修改 | `darr_mod_front/end/index/all` | 按cmp/key修改 |
| 排序 | `darr_sort_ascending/descending` | 升序/降序排列 |
| 存储 | `darr_store(handle, path)` | 二进制存储到文件 |
| 加载 | `darr_load(path)` | 从文件二进制加载 |
| 遍历 | `darr_travel(handle, op)` | 对每个元素执行op回调 |
| 统计 | `darr_num(handle)` | 返回数据个数 |
| 销毁 | `darr_destroy(&handle)` | 释放所有内存 |

### 3.12 设计要点总结

1. **void* 泛型**：通过 `handle->size` 记录单个元素大小，所有偏移计算都用 `index * size` 字节为单位
2. **每次插入重新 malloc**：当前实现每次插入都重新分配 `size * (num + 1)` 空间，简单但效率较低。生产环境可用 `realloc` + 容量预分配优化
3. **回调函数解耦**：通过函数指针（cmp/mod/sort/op）将数据操作与数据类型解耦，框架代码无需知道具体的结构体定义
4. **二级指针销毁**：`darr_destroy(DARR **)` 接受二级指针，可以在销毁后将外部指针置为 NULL，防止野指针

## 4. 单向链表

### 4.1 链表概述

**为什么需要链表：** 顺序表每次插入/删除都需要重新申请空间并拷贝全部数据，效率低。链表通过指针将分散的节点串联起来，插入/删除无需移动其他元素。

**单链表节点结构：**

```c
struct list_t {
    int data;              // 数据域：保存用户数据
    struct list_t *next;   // 指针域：指向下一个节点的地址
};
```

内存模型（非连续存储）：
```
[data|next] → [data|next] → [data|next] → NULL
  节点1          节点2          节点3
```

### 4.2 链表分类

链表按四个维度组合分类（2×2×2×2 = 16 种）：

| 维度 | 选项 | 说明 |
|------|------|------|
| 头节点 | 有头 / 无头 | 头节点不存数据，简化边界处理 |
| 排序 | 有序 / 无序 | 插入时是否按顺序排列 |
| 循环 | 循环 / 不循环 | 尾节点是否指向头节点 |
| 方向 | 单向 / 双向 | 节点有几个指针域 |

今天重点（星标为重要程度）：

| 类型 | 重要度 |
|------|--------|
| 有头 无序 不循环 单向 | **** |
| 无头 无序 不循环 单向 | *** |
| 有头 有序 不循环 单向 | 练习 |
| 无头 有序 不循环 单向 | 练习 |
| 有头 无序 循环 单向 | ** |

### 4.3 void* 指针偏移问题

`void*` 不能直接做算术运算，因为编译器不知道 `void` 的大小：

```c
void *v = NULL;
printf("%p\n", v + 1);       // 错误！void 无大小
printf("%p\n", (char *)v + 1); // 正确：按字节偏移
// 或按实际类型强转
printf("%p\n", ((struct cls_t *)v) + 1); // 偏移 sizeof(struct cls_t)
```

### 4.4 链表节点关系演示

```c
struct op_t {
    int data;
    struct op_t *p;
};

struct op_t a = {0, NULL};
struct op_t b = {20, NULL};
struct op_t c = {30, NULL};

a.p = &b;
b.p = &c;

printf("%d\n", a.p->data);     // 20 (a→b)
printf("%d\n", a.p->p->data);  // 30 (a→b→c)
```

### 4.5 有头 无序 不循环 单向链表 ★★★★

头节点在栈上分配，不保存数据，`head.next` 指向第一个实际节点。

**核心思路：** 找到最后一个节点 → 新节点挂在最后。

```c
// 头节点（不存数据，仅作为入口）
struct list_t head = {0, NULL};
struct list_t *new = NULL, *tail = NULL;
int num;

while (1) {
    scanf("%d", &num);
    if (num == -1) break;

    // 创建新节点
    new = (struct list_t *)malloc(sizeof(struct list_t));
    new->data = num;
    new->next = NULL;

    // 找最后一个节点
    for (tail = &head; tail->next != NULL; tail = tail->next)
        ;

    // 挂到尾部
    tail->next = new;
}

// 遍历：从 head.next 开始
for (tail = head.next; tail != NULL; tail = tail->next)
    printf("%d ", tail->data);

// 销毁：逐个释放
for (tail = head.next; tail != NULL; tail = new) {
    new = tail->next;
    free(tail);
}
```

**有头链表的优势：** 插入和删除操作统一，无需单独处理空链表的情况。

### 4.6 无头 无序 不循环 单向链表 ★★★

没有头节点，用 `head` 指针指向第一个节点。需要单独处理"链表为空"的情况。

```c
struct list_t *head = NULL;   // 空链表
struct list_t *new = NULL, *tail = NULL;
int num;

while (1) {
    scanf("%d", &num);
    if (num == -1) break;

    new = (struct list_t *)malloc(sizeof(struct list_t));
    new->data = num;
    new->next = NULL;

    if (head == NULL) {
        head = new;            // 第一个节点
    } else {
        // 找最后一个节点
        for (tail = head; tail->next != NULL; tail = tail->next)
            ;
        tail->next = new;
    }
}

// 遍历：从 head 开始
for (tail = head; tail != NULL; tail = tail->next)
    printf("%d ", tail->data);

// 销毁
for (tail = head; tail != NULL; tail = new) {
    new = tail->next;
    free(tail);
}
```

**有头 vs 无头对比：**

| 对比项 | 有头 | 无头 |
|--------|------|------|
| 头节点存储位置 | 栈上 | — |
| 空链表判断 | `head.next == NULL` | `head == NULL` |
| 首节点处理 | 统一操作 | 需单独判断 |
| 遍历起点 | `head.next` | `head` |
| 头节点不可释放 | ✓ | — |

### 4.7 有头 有序 不循环 单向链表

插入时按降序排列（大的在前）。在遍历找位置的同时比较大小，找到合适的插入点后执行**标准链表插入**。

```c
struct list_t head = {0, NULL};

while (1) {
    scanf("%d", &num);
    if (num == -1) break;

    new = (struct list_t *)malloc(sizeof(struct list_t));
    new->data = num;
    new->next = NULL;

    // 找插入位置：比 tail->next 大就停
    for (tail = &head; tail->next != NULL; tail = tail->next) {
        if (new->data > tail->next->data)
            break;
    }
    // 标准插入：新节点插入到 tail 和 tail->next 之间
    new->next = tail->next;
    tail->next = new;
}
```

**有序插入图示：**
```
插入前: head → [30] → [20] → [10] → NULL
插入 25:
  找到 tail 指向 [30]（因为 25 < 30->data）
  new->next = tail->next   → new 指向 [30]
  tail->next = new         → [20]→next = new
结果: head → [30] → [25] → [20] → [10] → NULL
              ↑       ↑
             tail     new
```

### 4.8 无头 有序 不循环 单向链表

比有头版本多了两个边界判断：链表为空、插入位置在头部。

```c
struct list_t *head = NULL;

while (1) {
    scanf("%d", &num);
    if (num == -1) break;

    new = (struct list_t *)malloc(sizeof(struct list_t));
    new->data = num;
    new->next = NULL;

    if (head == NULL) {
        head = new;                    // 空链表：直接作为首个节点
    } else if (new->data > head->data) {
        new->next = head;              // 头部插入
        head = new;
    } else {
        for (tail = head; tail->next != NULL; tail = tail->next) {
            if (new->data > tail->next->data)
                break;
        }
        new->next = tail->next;        // 中间插入
        tail->next = new;
    }
}
```

**三种情况的判断顺序：**
1. 链表为空 → 直接当头节点
2. 新数据 > 头数据 → 头部插入（更新 head）
3. 其他情况 → 中间/尾部插入

### 4.9 有头 无序 循环 单向链表 ★★

尾节点的 `next` 指向头节点（`&head`），形成环。判断结束的条件从 `!= NULL` 变为 `!= &head`。

```c
// 头节点的 next 指向自己，形成自循环
struct list_t head = {0, &head};

while (1) {
    scanf("%d", &num);
    if (num == -1) break;

    new = (struct list_t *)malloc(sizeof(struct list_t));
    new->data = num;
    new->next = NULL;

    // 找最后一个节点（tail->next == &head 说明到了末尾）
    for (tail = &head; tail->next != &head; tail = tail->next)
        ;

    // 新节点指向头，尾节点指向新节点
    new->next = &head;
    tail->next = new;
}

// 遍历：直到回到 head
for (tail = head.next; tail != &head; tail = tail->next)
    printf("%d ", tail->data);

// 销毁：直到回到 head
for (tail = head.next; tail != &head; tail = new) {
    new = tail->next;
    free(tail);
}
```

**循环链表 vs 不循环链表：**

| 对比项 | 不循环 | 循环 |
|--------|--------|------|
| 头节点初始化 | `{0, NULL}` | `{0, &head}` |
| 尾节点 next | `NULL` | `&head` |
| 遍历结束条件 | `tail != NULL` | `tail != &head` |
| 找尾节点条件 | `tail->next != NULL` | `tail->next != &head` |
| 新节点 next | `NULL` | `&head` |

### 4.10 链表反转（有头 无序 不循环）

反转单链表的经典算法：遍历链表，逐个将节点的 `next` 指针反向。

```c
struct list_t *save = NULL, *prev = NULL;

// tail 从第一个节点开始
tail = head.next;
while (tail != NULL)
{
    save = tail->next;   // 1. 保存下一个节点
    tail->next = prev;   // 2. 反转当前节点的 next 指针
    prev = tail;         // 3. prev 前移到当前节点
    tail = save;         // 4. tail 前移到保存的下一个节点
}
head.next = prev;        // 更新头节点指向新的第一个节点
```

**反转过程图示：**
```
原链: head → [A] → [B] → [C] → NULL

第1步: NULL ← [A]    [B] → [C] → NULL   (prev=A, save=B)
第2步: NULL ← [A] ← [B]    [C] → NULL   (prev=B, save=C)
第3步: NULL ← [A] ← [B] ← [C]           (prev=C, save=NULL)

结果: head → [C] → [B] → [A] → NULL
```

**关键点：** 必须先用 `save` 保存 `tail->next`，否则断了后无法继续遍历。

### 4.11 无头 无序 循环 单向链表

```c
while (1) {

// 遍历：tail->next != head 时继续，最后再打印最后一个节点

// 销毁
```

### 4.12 有头 有序 循环 单向链表

```c
struct list_t head = {0, &head};

while (1) {

    // 找插入位置（降序），循环 + 有序的合并条件
    for (tail = &head; tail->next != &head && new->data > tail->next->data; tail = tail->next)
```

### 4.13 无头 有序 循环 单向链表

相比有头版本，多了"链表为空"和"头部插入（升序时是最小值在头）"的判断：

```c
struct list_t *head = NULL;

while (1) {

        head = new;
```

## 5. 双向链表

### 5.1 节点结构

```c
struct list_t {
    int data;              // 数据域
    struct list_t *next;   // 指向下一个节点
    struct list_t *prev;   // 指向上一个节点
};
```

### 5.2 有头 无序 循环 双向链表 ★★★★★

**头节点初始化：** `struct list_t head = {0, &head, &head};` — next 和 prev 都指向自己。

**核心插入算法（尾插法）：**

```c
struct list_t head = {0, &head, &head};

while (1) {
    scanf("%d", &num);
    if (num == -1) break;

    new = (struct list_t *)malloc(sizeof(struct list_t));
    new->data = num;
    new->next = NULL;
    new->prev = NULL;

    // 四步插入法（"先写外来的，再谁长写谁"）
    new->next = &head;              // 1. 新节点 next 指向头
    new->prev = head.prev;          // 2. 新节点 prev 指向原来的尾节点
    head.prev->next = new;          // 3. 原尾节点的 next 指向新节点
    head.prev = new;                // 4. 头节点的 prev 指向新节点
}
```

**四步插入图示：**
```
插入前: head ⇄ [A] ⇄ [B]          (head.prev = B, B.next = head)
                     ↑
                   尾节点

插入 new 后:
  1. new->next = &head     new 的后继是头
  2. new->prev = head.prev  new 的前驱是原尾节点 B
  3. B.next = new           B 的后继变为 new
  4. head.prev = new        头的"前驱"（即尾节点）变为 new

结果: head ⇄ [A] ⇄ [B] ⇄ [new]
```

**正向遍历（next）：**
```c
for (tail = head.next; tail != &head; tail = tail->next)
    printf("%d ", tail->data);
```

**反向遍历（prev）：**
```c
for (tail = head.prev; tail != &head; tail = tail->prev)
    printf("%d ", tail->data);
```

**口诀：** "先写外来的数据（new 的两个指针），再谁长写谁（调整相邻节点的指针）"

### 5.3 无头 无序 循环 双向链表 ★★★★

```c
struct list_t *head = NULL;

while (1) {
    // ... 创建 new ...
    if (head == NULL) {
        new->next = new;     // 自循环
        new->prev = new;
        head = new;
    }
    // 尾插（同有头版本，但用 head 替代 &head）
    new->next = head;
    new->prev = head->prev;
    head->prev->next = new;
    head->prev = new;
}

// 正向遍历
for (tail = head; tail->next != head; tail = tail->next)
    printf("%d ", tail->data);
printf("%d\n", tail->data);

// 反向遍历
for (tail = head->prev; tail != head; tail = tail->prev)
    printf("%d ", tail->data);
printf("%d\n", tail->data);
```

### 5.4 有头 有序 循环 双向链表

```c
struct list_t head = {0, &head, &head};

while (1) {
    // ... 创建 new ...
    // 找插入位置（降序）
    for (tail = &head; tail->next != &head && new->data > tail->next->data; tail = tail->next)
        ;

    // 四步插入（tail 和 tail->next 之间）
    new->next = tail->next;
    new->prev = tail;
    tail->next->prev = new;
    tail->next = new;
}
```

### 5.5 无头 有序 循环 双向链表

比有头版本多两个边界处理：

```c
struct list_t *head = NULL;

while (1) {
    // ... 创建 new ...
    if (head == NULL) {
        new->next = new;
        new->prev = new;
        head = new;
    }
    if (new->data < head->data) {  // 升序：头部插入
        new->next = head;
        new->prev = head->prev;
        head->prev->next = new;
        head->prev = new;
        head = new;                 // 更新头指针
    } else {
        for (tail = head; tail->next != head && new->data > tail->next->data; tail = tail->next)
            ;
        new->next = tail->next;
        new->prev = tail;
        tail->next->prev = new;
        tail->next = new;
    }
}
```

### 5.6 双向链表插入通用模式

无论是头插、尾插、中间插，四步操作统一为：

```c
// 在 tail 后面插入 new
new->next = tail->next;      // 1. new 的 next
new->prev = tail;            // 2. new 的 prev
tail->next->prev = new;      // 3. tail->next 的 prev
tail->next = new;            // 4. tail 的 next
```

**删除节点通用模式：**
```c
// 删除 tail 指向的节点
tail->prev->next = tail->next;
tail->next->prev = tail->prev;
free(tail->data);    // 如有独立数据空间
free(tail);
```

## 6. 链表框架封装（LLIST）

仿照 DARR 的设计模式，将 **有头 无序 循环 双向链表** 封装为通用框架。

### 6.1 框架结构

```c
// 双向链表节点
struct node_t {
    void *data;            // 数据域（void*泛型）
    struct node_t *next;   // 后继指针
    struct node_t *prev;   // 前驱指针
};

// 链表句柄（有头 无序 循环 双向）
typedef struct llist_t {
    struct node_t head;    // 头节点（不存数据）
    int size;              // 数据类型大小
    int num;               // 数据个数
} LLIST;
```

**回调函数类型（与 DARR 一致）：**
```c
typedef void (llist_op_t)(const void *);
typedef int  (llist_cmp_t)(const void *, const void *);
typedef void (llist_mod_t)(void *, const void *);
typedef int  (llist_sort_t)(const void *, const void *);
```

### 6.2 创建与初始化

```c
LLIST *llist_creat(int size)
{
    LLIST *handle = (LLIST *)malloc(sizeof(LLIST));

    handle->size = size;
    handle->num = 0;
    handle->head.data = NULL;
    // 循环双向：next 和 prev 都指向自己
    handle->head.next = &handle->head;
    handle->head.prev = &handle->head;

    return handle;
}
```

### 6.3 插入操作

链表节点需要两次 malloc：一次为节点结构体，一次为节点中的数据。

```c
int llist_insert_end(LLIST *handle, void *data)
{
    struct node_t *new = (struct node_t *)malloc(sizeof(struct node_t));
    new->data = malloc(handle->size);
    memcpy(new->data, data, handle->size);

    // 尾插 = 在头节点前面插入
    new->next = &handle->head;
    new->prev = handle->head.prev;
    handle->head.prev->next = new;
    handle->head.prev = new;

    handle->num++;
    return 0;
}

int llist_insert_front(LLIST *handle, void *data)
{
    // 头插 = 在头节点后面插入
    // new->next = handle->head.next;
    // new->prev = &handle->head;
    // handle->head.next->prev = new;
    // handle->head.next = new;
}

int llist_insert_index(LLIST *handle, void *data, int index)
{
    // 定位到 index 位置的节点，在其前插入
    struct node_t *tail = &handle->head;
    while (index--) tail = tail->next;
    // 四步插入...
}
```

### 6.4 删除操作（仅调整指针，不需要拷贝数据）

```c
void llist_del_front(LLIST *handle, void *key, llist_cmp_t *cmp)
{
    struct node_t *tail;
    for (tail = handle->head.next; tail != &handle->head; tail = tail->next)
    {
        if (cmp(tail->data, key) == 0)
        {
            tail->prev->next = tail->next;
            tail->next->prev = tail->prev;
            free(tail->data);   // 先释放数据
            free(tail);         // 再释放节点
            handle->num--;
            return;
        }
    }
}
```

**链表 vs 顺序表删除的核心区别：**
- 顺序表：需要重新 malloc 并拷贝所有数据
- 链表：只调整相邻节点的指针，效率 O(1)

### 6.5 排序操作

不交换节点，只交换节点中的 `data` 指针（比移动节点更简单）：

```c
typedef enum { ASCENDING, DESCENDING } SORT;

void llist_sort(LLIST *handle, llist_sort_t *sort, SORT flag)
{
    struct node_t *p1, *p2;
    void *tmp;

    for (p1 = handle->head.next; p1 != &handle->head; p1 = p1->next)
    {
        for (p2 = p1->next; p2 != &handle->head; p2 = p2->next)
        {
            if (flag == ASCENDING && sort(p1->data, p2->data) > 0)
            {
                tmp = p1->data;           // 只交换数据指针
                p1->data = p2->data;
                p2->data = tmp;
            }
        }
    }
}
```

### 6.6 存储与加载

```c
// 存储格式: [size][num][data1][data2]...
void llist_store(LLIST *handle, const char *path)
{
    FILE *fp = fopen(path, "w");
    fwrite(&handle->size, sizeof(handle->size), 1, fp);
    fwrite(&handle->num, sizeof(handle->num), 1, fp);
    for (tail = handle->head.next; tail != &handle->head; tail = tail->next)
        fwrite(tail->data, handle->size, 1, fp);
    fclose(fp);
}

// 加载：读 size → 创建 handle → 逐个读数据并插入
LLIST *llist_load(const char *path);
```

### 6.7 正向/反向遍历

```c
void llist_travel_next(LLIST *handle, llist_op_t *op)
{
    struct node_t *tail;
    for (tail = handle->head.next; tail != &handle->head; tail = tail->next)
        op(tail->data);
}

void llist_travel_prev(LLIST *handle, llist_op_t *op)
{
    struct node_t *tail;
    for (tail = handle->head.prev; tail != &handle->head; tail = tail->prev)
        op(tail->data);
}
```

### 6.8 销毁

```c
void llist_destroy(LLIST **handle)
{
    struct node_t *tail, *save;

    for (tail = (*handle)->head.next; tail != &(*handle)->head; tail = save)
    {
        save = tail->next;
        free(tail->data);   // 先释放节点中的数据
        free(tail);         // 再释放节点本身
    }
    free(*handle);
    *handle = NULL;
}
```

| 分类 | 函数 | 说明 |
|------|------|------|
| 创建 | `llist_creat(size)` | 创建有头无序循环双向链表 |

### 6.10 链表 vs 顺序表总结

| 对比项 | 顺序表 (DARR) | 链表 (LLIST) |
|--------|---------------|--------------|
| 内存布局 | 连续空间 | 分散节点 |
| 插入效率 | O(n) — 重新分配+拷贝全部 | O(n) 查找 + O(1) 插入 |
| 删除效率 | O(n) — 重新分配+拷贝全部 | O(n) 查找 + O(1) 删节点 |
| 随机访问 | O(1) — 直接下标 | O(n) — 需遍历 |
| 节点结构 | 只存数据 | 数据 + 2个指针 |
| 内存开销 | 小（仅有数据） | 大（每个节点多2个指针） |
| 双向遍历 | 不支持 | 支持 |

## 7. 栈 (Stack)

### 7.1 基本概念

**特性：** 先进后出（FILO: First In Last Out），后进先出（LIFO）。

**类比：** 手枪弹夹 — 先压入的子弹最后打出。

**核心操作：**
| 操作 | 说明 |
|------|------|
| 入栈 (push) | 数据压入栈顶 |
| 出栈 (pop) | 从栈顶取出数据 |
| 满栈判断 | `top == max` |
| 空栈判断 | `top == 0` |

### 7.2 数组实现

```c
#define MAX 10

int is_full(int top)  { return top == MAX; }
int is_empty(int top) { return top == 0; }

// 入栈
int push(int *stack, int num, int *top)
{
    if (is_full(*top)) return -1;   // 满栈
    stack[*top] = num;
    (*top)++;
    return 0;
}

// 出栈
int pop(int *stack, int *top)
{
    if (is_empty(*top)) return -1;   // 空栈
    (*top)--;
    return stack[*top];
}
```

### 7.3 泛型栈封装（基于动态数组）

将栈封装为通用框架，结构与 DARR 类似：

```c
typedef struct stack_t {
    void *stack;   // 栈空间
    int size;      // 数据类型大小
    int max;       // 最大容量
    int top;       // 栈顶指针
} STACK;

// 创建
STACK *stack_creat(int size, int max);

// 入栈
void stack_push(STACK *handle, void *data)
{
    if (stack_is_full(handle)) return;
    memcpy(handle->stack + handle->top * handle->size, data, handle->size);
    handle->top++;
}

// 出栈：返回数据地址（指针仍指向栈内空间，读取后失效）
void *stack_pop(STACK *handle)
{
    if (stack_is_empty(handle)) return NULL;
    handle->top--;
    return handle->stack + handle->top * handle->size;
}

// 清空栈（只需重置 top）
void stack_clean(STACK *handle) { handle->top = 0; }
```

**存储格式：** `[size][max][top][data...]`

### 7.4 完整示例：int 栈数组实现

```c
#include <stdio.h>
#include <stdlib.h>

#define MAX 10

int is_full(int top)  { return top == MAX; }
int is_empty(int top) { return top == 0; }

int push(int *stack, int num, int *top)
{
    if (is_full(*top)) {
        printf("栈满，%d 入栈失败\n", num);
        return -1;
    }
    stack[*top] = num;
    (*top)++;
    printf("%d 入栈, top = %d\n", num, *top);
    return 0;
}

int pop(int *stack, int *top)
{
    if (is_empty(*top)) {
        printf("栈空，出栈失败\n");
        return -1;
    }
    (*top)--;
    int val = stack[*top];
    printf("%d 出栈, top = %d\n", val, *top);
    return val;
}

void travel(int *stack, int top)
{
    printf("栈内元素(底→顶): ");
    for (int i = 0; i < top; i++)
        printf("%d ", stack[i]);
    printf("\n");
}

int main(void)
{
    int stack[MAX];
    int top = 0;

    // 入栈 MAX+2 个元素（最后2个会满）
    for (int i = 0; i < MAX + 2; i++) {
        int num = rand() % 100;
        push(stack, num, &top);
    }
    travel(stack, top);

    // 出栈 3 个
    printf("\n--- 出栈 3 个 ---\n");
    pop(stack, &top);
    pop(stack, &top);
    pop(stack, &top);
    travel(stack, top);

    // 再入栈 2 个
    printf("\n--- 再入栈 2 个 ---\n");
    push(stack, 100, &top);
    push(stack, 200, &top);
    travel(stack, top);

    // 全部出栈
    printf("\n--- 全部出栈 ---\n");
    while (!is_empty(top))
        pop(stack, &top);

    return 0;
}
```

```
--- 出栈 3 个 ---
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// ===== 栈框架（精简版）=====
typedef struct stack_t {
    void *stack;   int size;   int max;   int top;
} STACK;

STACK *stack_creat(int size, int max)
{
    STACK *h = (STACK *)malloc(sizeof(STACK));
    h->stack = malloc(max * size);
    h->size = size; h->max = max; h->top = 0;
    return h;
}
int  stack_is_full(STACK *h)  { return h->top == h->max; }
int  stack_is_empty(STACK *h) { return h->top == 0; }
void stack_push(STACK *h, void *data) {
    if (stack_is_full(h)) return;
    memcpy(h->stack + h->top * h->size, data, h->size);
    h->top++;
}
void *stack_pop(STACK *h) {
    if (stack_is_empty(h)) return NULL;
    h->top--;
    return h->stack + h->top * h->size;
}
void stack_destroy(STACK **h) {
    free((*h)->stack); free(*h); *h = NULL;
}

// ===== 学生结构体 =====
struct cls_t {
    char name[64]; int id; int age; float sum;
};

void ls(const void *data)
{
    const struct cls_t *c = data;
    printf("name:%s id:%d age:%d sum:%.2f\n", c->name, c->id, c->age, c->sum);
}

int main(void)
{
    STACK *handle = stack_creat(sizeof(struct cls_t), 8);
    struct cls_t cls, *p;

    // 入栈 8 个学生（随机生成）
    for (int i = 0; i < 8; i++) {
        snprintf(cls.name, sizeof(cls.name), "stu_%c%c",
                 rand() % 26 + 'A', rand() % 26 + 'a');
        cls.id = 1001 + i;
        cls.age = 18 + rand() % 5;
        cls.sum = 60.0 + (rand() % 400) / 10.0;
        stack_push(handle, &cls);
    }

    printf("===== 全部入栈后 =====\n");
    for (int i = 0; i < handle->top; i++)
        ls((struct cls_t *)handle->stack + i);

    printf("\n===== 出栈 3 个（后进先出）=====\n");
    for (int i = 0; i < 3; i++) {
        p = (struct cls_t *)stack_pop(handle);
        if (p) { printf("pop -> "); ls(p); }
    }

    printf("\n===== 剩余栈内 =====\n");
    for (int i = 0; i < handle->top; i++)
        ls((struct cls_t *)handle->stack + i);

    stack_destroy(&handle);
    return 0;
}
```

### 7.7 应用示例：逆波兰表达式求值

```c
#include <stdio.h>

int main(void)
{
    char expr[] = "23+5*";        // (2+3)*5 = 25
    int stack[64], top = 0;

    printf("逆波兰表达式: %s\n", expr);

    for (int i = 0; expr[i] != '\0'; i++) {
        char c = expr[i];
        if (c >= '0' && c <= '9') {
            stack[top++] = c - '0';
            printf("push %d\n", c - '0');
        } else {
            int b = stack[--top];  // 右操作数（后弹出）
            int a = stack[--top];  // 左操作数（先弹出）
            int r;
            switch (c) {
                case '+': r = a + b; break;
                case '-': r = a - b; break;
                case '*': r = a * b; break;
                case '/': r = a / b; break;
            }
            stack[top++] = r;
            printf("%d %c %d = %d\n", a, c, b, r);
        }
    }

    printf("结果: %d\n", stack[0]);
    return 0;
}
```

```
3           → stack: [2, 3]
+           → 2+3=5, stack: [5]
5           → stack: [5, 5]
*           → 5*5=25, stack: [25]
结果: 25
```

## 8. 队列 (Queue)

### 8.1 基本概念

**特性：** 先进先出（FIFO: First In First Out），后进后出。

**类比：** 排队买票 — 先来的先服务。

### 8.2 循环队列（数组实现）

用数组实现循环队列，牺牲一个位置来区分满和空：

```c
#define MAX 10

int is_full(int front, int end)  { return (end + 1) % MAX == front; }
int is_empty(int front, int end) { return end == front; }

// 入队
void qe(int *queue, int num, int front, int *end)
{
    if (is_full(front, *end)) return;
    queue[*end] = num;
    (*end)++;
    if (*end == MAX) *end = 0;   // 循环
}

// 出队
int de(int *queue, int *front, int end)
{
    if (is_empty(*front, end)) return -1;
    int save = queue[*front];
    (*front)++;
    if (*front == MAX) *front = 0; // 循环
    return save;
}

// 队列长度
int len(int front, int end)
{
    if (end >= front)
        return end - front;
    else
        return MAX - (front - end);
}
```

### 8.3 完整示例：循环队列 int 实现

```c
#include <stdio.h>
#include <stdlib.h>

#define MAX 10

int is_full(int front, int end)  { return (end + 1) % MAX == front; }
int is_empty(int front, int end) { return end == front; }

void qe(int *queue, int num, int front, int *end)
{
    if (is_full(front, *end)) {
        printf("队满，%d 入队失败\n", num);
        return;
    }
    queue[*end] = num;
    printf("%d 入队 (end=%d)\n", num, *end);
    (*end)++;
    if (*end == MAX) *end = 0;
}

int de(int *queue, int *front, int end)
{
    if (is_empty(*front, end)) {
        printf("队空，出队失败\n");
        return -1;
    }
    int save = queue[*front];
    printf("%d 出队 (front=%d)\n", save, *front);
    (*front)++;
    if (*front == MAX) *front = 0;
    return save;
}

int qlen(int front, int end)
{
    return (end >= front) ? (end - front) : (MAX - (front - end));
}

void travel(int *queue, int front, int end)
{
    printf("队列内容(队头→队尾): ");
    for (int i = 0, j = front; i < qlen(front, end); i++, j++) {
        if (j == MAX) j = 0;
        printf("%d ", queue[j]);
    }
    printf("  长度=%d\n", qlen(front, end));
}

int main(void)
{
    int queue[MAX];
    int front = 0, end = 0;

    // 入队 MAX 个（最后一个会满）
    printf("===== 入队 10 个（实际入队 9 个，牺牲 1 位）=====\n");
    for (int i = 0; i < MAX; i++) {
        int num = rand() % 100;
        qe(queue, num, front, &end);
    }
    travel(queue, front, end);

    // 出队 3 个
    printf("\n===== 出队 3 个（先进先出）=====\n");
    for (int i = 0; i < 3; i++)
        de(queue, &front, end);
    travel(queue, front, end);

    // 再入队（利用循环空间）
    printf("\n===== 再入队 3 个（循环利用空间）=====\n");
    qe(queue, 100, front, &end);
    qe(queue, 200, front, &end);
    qe(queue, 300, front, &end);
    travel(queue, front, end);

    // 全部出队
    printf("\n===== 全部出队 =====\n");
    while (!is_empty(front, end))
        de(queue, &front, end);
    travel(queue, front, end);

    return 0;
}
```

**运行效果：**
```

83 入队 (end=0)
77 入队 (end=1)
...
队满，27 入队失败          ← 第10个失败
队列内容: 83 77 15 93 35 86 49 21 62   长度=9

===== 出队 3 个（先进先出）=====
83 出队 (front=0)          ← 最先入队的最先出（FIFO）
77 出队 (front=1)
15 出队 (front=2)
队列内容: 93 35 86 49 21 62   长度=6

===== 再入队 3 个（循环利用空间）=====
100 入队 (end=9)
200 入队 (end=0)           ← end 循环回 0，复用空位
300 入队 (end=1)
队列内容: 93 35 86 49 21 62 100 200 300   长度=9

===== 全部出队 =====
93 出队 (front=3)
...
300 出队 (front=1)
队空，出队失败
队列内容:   长度=0
```

### 8.4 循环队列核心要点

**为什么 `(end + 1) % MAX == front` 是满队条件：**
- 当队尾追上队头时即为满，但直接 `end == front` 无法区分空和满
- 牺牲一个位置：保留一个空位，使得满队时 `end` 在 `front` 前一个位置

**空队/满队区分：**
```
空队: front == end
      [ ][ ][ ][ ][ ]
      f/e

满队: (end + 1) % MAX == front （实际容量 = MAX - 1）
      [X][X][X][ ][X]
          e   f
      end在front前一个位置，留了一个空位
```

**队列长度计算：**
```c
// end 在 front 后面（正常情况）：直接相减
// end 在 front 前面（循环后）：总长 - 间隔
int len = (end >= front) ? (end - front) : (MAX - (front - end));
```

## 9. 排序算法

### 9.1 概述

排序是将无序数据按一定规则（升序/降序）重新排列的过程。常用排序算法按时间复杂度分档：

| 算法 | 时间复杂度 | 空间 | 稳定 |
|------|-----------|------|------|
| 冒泡排序 | O(n²) | O(1) | 是 |
| 选择排序 | O(n²) | O(1) | 否 |
| 插入排序 | O(n²) | O(1) | 是 |
| 交换排序 | O(n²) | O(1) | 否 |
| 快速排序 | O(n log n) | O(log n) | 否 |

### 9.2 交换排序

取出一个数据，跟后面所有的数据比较，满足条件则交换。

```c
void swap_sort(int *arr, int n)
{
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (arr[i] > arr[j]) {      // 升序：大的往后换
                int tmp = arr[i];
                arr[i] = arr[j];
                arr[j] = tmp;
            }
        }
    }
}
```

**过程（升序）：**
```
原始: [83, 86, 77, 15, 93]
i=0:  [15, 86, 83, 77, 93]  ← 83与77,15交换，最小的15到位置0
i=1:  [15, 77, 86, 83, 93]  ← 86与77交换
i=2:  [15, 77, 83, 86, 93]  ← 86与83交换
i=3:  [15, 77, 83, 86, 93]  ← 已排序
```

### 9.3 冒泡排序

相邻两个数比较，小数上浮（前移），大数下沉（后移）。每轮将最大的数"冒泡"到最后。

```c
void bubble_sort(int *arr, int n)
{
    for (int i = 0; i < n; i++) {
        // 每轮将最大值冒泡到最后 n-i-1 位置
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                // XOR 交换（无需临时变量）
                arr[j] ^= arr[j + 1];
                arr[j + 1] ^= arr[j];
                arr[j] ^= arr[j + 1];
            }
        }
    }
}
```

**过程图示：**
```
原始:  [83, 86, 77, 15, 93]

第1轮（i=0）：比较相邻，大者右移
  83<86 ✓ → [83, 86, 77, 15, 93]
  86>77 ✗ → [83, 77, 86, 15, 93]  交换
  86>15 ✗ → [83, 77, 15, 86, 93]  交换
  86<93 ✓ → [83, 77, 15, 86, 93]
  结果: 93 沉底 ✓

第2轮（i=1）：→ [77, 15, 83, 86, 93]
第3轮（i=2）：→ [15, 77, 83, 86, 93]
第4轮（i=3）：→ [15, 77, 83, 86, 93]  ← 已有序
```

**XOR 交换原理：**
```c
a = a ^ b;    // a = a⊕b
b = b ^ a;    // b = b⊕(a⊕b) = a
a = a ^ b;    // a = (a⊕b)⊕a = b
// 注意：a 和 b 不能是同一变量，否则会清零
```

### 9.4 选择排序

每轮找到最小值，把它放到最前面。比交换排序减少了交换次数。

```c
void select_sort(int *arr, int n)
{
    for (int i = 0; i < n; i++) {
        int min = i;                    // 假设当前位置最小
        for (int j = i + 1; j < n; j++) {
            if (arr[min] > arr[j]) {
                min = j;                // 记录更小值的位置
            }
        }
        if (min != i) {                 // 只交换一次
            int tmp = arr[i];
            arr[i] = arr[min];
            arr[min] = tmp;
        }
    }
}
```

**选择排序 vs 交换排序：** 选择排序每轮只交换1次（找到最小值后），交换排序每次找到更小值就交换，所以选择排序的交换次数更少。

### 9.5 插入排序

从第二个元素开始，取出当前值，与前面已排序部分比较，找到合适位置插入。类似打牌时整理手牌。

```c
void insert_sort(int *arr, int n)
{
    for (int i = 1; i < n; i++) {
        int save = arr[i];             // 取出当前值
        int j;
        for (j = i; j > 0; j--) {
            if (save < arr[j - 1]) {
                arr[j] = arr[j - 1];   // 前一个值后移
            } else {
                break;                  // 找到位置
            }
        }
        arr[j] = save;                  // 放入正确位置
    }
}
```

**过程图示：**
```
原始: [83, 86, 77, 15, 93]

i=1: save=86, 86>83 → break → [83, 86, 77, 15, 93]
i=2: save=77, 77<86 → arr[2]=86 → 77<83 → arr[1]=83 → [77, 83, 86, 15, 93]
i=3: save=15, 15<86 → arr[3]=86 → 15<83 → arr[2]=83 → 15<77 → arr[1]=77
     → arr[0]=15 → [15, 77, 83, 86, 93]
i=4: save=93, 93>86 → break → [15, 77, 83, 86, 93]
```

**时间复杂度分析：** 最好 O(n)（已有序），最坏 O(n²)（逆序）。适合小规模或基本有序的数据。

### 9.6 C库快速排序 — qsort()

```c
#include <stdlib.h>

           int (*compar)(const void *, const void *));
```

| 参数 | 含义 |
|------|------|
| `base` | 数据存储空间地址 |
| `nmemb` | 数据个数 |
| `size` | 每个数据的类型大小 |
| `compar` | 比较函数指针 |

**int 比较函数：**
```c
int cmp_int(const void *d1, const void *d2)
{
    return *(int *)d1 - *(int *)d2;   // 升序
    // return *(int *)d2 - *(int *)d1; // 降序
}
```

**float 比较函数（注意精度）：**
```c

{
    float f = *(float *)d1 - *(float *)d2;
    if (f > 0.00001)  return 1;
    if (f < -0.00001) return -1;
    return 0;
}
```

**使用示例：**
```c

```

### 9.7 自实现快速排序（递归）

```c
void my_qsort(int *arr, int l, int h)
{
    if (l > h) return;                // 递归终止条件

    int base = arr[l];                // 基准值
    int low = l;                      // 保存边界用于递归
    int high = h;

    while (l < h) {
        // 从高位开始：找比 base 小的
        while (base < arr[h] && l < h)
            h--;
        if (l < h) {
            arr[l] = arr[h];          // 高位小数移到低位
            l++;
        }

        // 从低位开始：找比 base 大的
        while (base > arr[l] && l < h)
            l++;
        if (l < h) {
            arr[h] = arr[l];          // 低位大数移到高位
            h--;
        }
    }
    arr[l] = base;                    // 基准归位

    my_qsort(arr, low, l - 1);        // 递归左半
    my_qsort(arr, l + 1, high);       // 递归右半
}
```

**关键细节：**
- 递归终止条件为 `l > h`（而非 `l >= h`），与保存的 `low`/`high` 配合
- 必须先保存 `low`/`high`，因为 `l`/`h` 在循环中会变化
- 从高位开始找小数，从低位开始找大数，交替进行

**执行过程（`[83, 86, 77, 15, 93, 35, 86, 92, 49, 21]`）：**
```
base=83:
  h从21开始: 21<83 → arr[0]=21, l++
  l从86开始: 86>83 → arr[9]=86, h--
  h从49开始: 49<83 → arr[1]=49, l++
  l从77开始: 77<83 → l++... 直到86>83 → arr[8]=86, h--
  ...
  l==h → arr[l]=83 (基准归位)
  → 左边都<83, 右边都>83

递归左右: my_qsort(arr, 0, l-1), my_qsort(arr, l+1, 9)
```

## 10. 查找算法

### 10.1 顺序查找

从头到尾逐个比较，时间复杂度 O(n)：

```c
int find(int *arr, int n, int key)
{
    for (int i = 0; i < n; i++)
        if (arr[i] == key)
            return arr[i];
    return -1;
}
```

### 10.2 二分查找

**前提：** 数据必须有序。每次取中间值比较，排除一半数据，时间复杂度 O(log n)。

```c
int my_bsearch(int *arr, int l, int h, int key)
{
    while (l <= h) {
        int m = (l + h) / 2;        // 中间位置
        if (key > arr[m])
            l = m + 1;              // 去右半找
        else if (key < arr[m])
            h = m - 1;              // 去左半找
        else
            return arr[m];          // 找到了
    }
    return -1;                      // 没找到
}
```

**查找过程（在 `[15, 35, 49, 77, 83, 86, 92, 93]` 中找 86）：**
```
第1次: l=0 h=7 m=3 arr[3]=77  → 86>77, l=4
第2次: l=4 h=7 m=5 arr[5]=86  → 找到! 返回 86
```

**注意终止条件：** `l <= h`（不是 `l < h`），否则当 `l==h` 且该位置就是目标时会漏掉。

### 10.3 C库二分查找 — bsearch()

```c
#include <stdlib.h>

void *bsearch(const void *key, const void *base,
              size_t nmemb, size_t size,
              int (*compar)(const void *, const void *));
```

| 参数 | 含义 |
|------|------|
| `key` | 查找的关键字地址 |
| `base` | 数据存储空间地址（必须有序） |
| `nmemb` | 数据个数 |
| `size` | 数据类型大小 |
| `compar` | 比较函数指针 |
| 返回值 | NULL 表示未找到，否则返回匹配元素地址 |

```c
int arr[] = {15, 35, 49, 77, 83, 86, 92, 93};
int key = 86;
int *p = (int *)bsearch(&key, arr, 8, sizeof(int), cmp_int);
if (p != NULL)
    printf("find: %d\n", *p);     // find: 86
```

**注意：** bsearch 要求数据预先排序，否则结果不可预测。

## 11. 哈希表 (Hash Table)

### 11.1 基本概念

**问题：** 顺序查找太慢 O(n)，二分查找需要先排序 O(n log n)。

**哈希思想：** 通过哈希函数将 key 映射为数组下标，直接定位，理想情况 O(1)。

**冲突处理（拉链法）：** 不同 key 映射到同一位置时，用链表串起来。

### 11.2 节点结构

```c
struct hash_t {
    int data;              // 数据域
    struct hash_t *next;   // 指针域（拉链）
};
```

哈希表 = 数组 + 链表：
```
index 0: head[0] → [30] → [20] → NULL    ← 30%10=0, 20%10=0
index 1: head[1] → [1] → NULL            ← 1%10=1
index 2: head[2] → NULL
...
index 9: head[9] → NULL
```

### 11.3 哈希函数

```c
int hash(int num)
{
    return num % MAX;  // 取模法，MAX 为表大小
}
```

常见的哈希函数设计：
| 方法 | 公式 | 适用 |
|------|------|------|
| 取模法 | `key % MAX` | 整数（最常用） |
| 平方取中 | 取 key² 的中间几位 | 整数 |
| 折叠法 | 将 key 分段求和 | 长整数 |

### 11.4 完整实现（有头 单向 不循环 哈希）

```c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

#define MAX 10

struct hash_t {
    int data;
    struct hash_t *next;
};

int hash(int num) { return num % MAX; }

int main(void)
{
    int num, key, index;
    struct hash_t head[MAX] = {};  // 头节点数组（全部初始化为0）
    struct hash_t *new = NULL, *tail = NULL;

    srand(time(NULL));

    // ===== 1. 构建哈希表 =====
    for (int i = 0; i < MAX; i++) {
        num = rand() % 100;
        printf("%d ", num);

        new = (struct hash_t *)malloc(sizeof(struct hash_t));
        new->data = num;
        new->next = NULL;

        index = hash(num);          // 计算哈希值

        // 找到对应桶的尾部，插入
        for (tail = &head[index]; tail->next != NULL; tail = tail->next)
            ;
        tail->next = new;
    }
    printf("\n");

    // ===== 2. 遍历哈希表 =====
    for (int i = 0; i < MAX; i++) {
        printf("%d : ", i);
        for (tail = head[i].next; tail != NULL; tail = tail->next)
            printf("%d ", tail->data);
        printf("\n");
    }

    // ===== 3. 哈希查找 =====
    printf("input find key: ");
    scanf("%d", &key);
    index = hash(key);
    for (tail = head[index].next; tail != NULL; tail = tail->next) {
        if (tail->data == key) {
            printf("find: %d (in bucket %d)\n", tail->data, index);
        }
    }

    // ===== 4. 销毁哈希表 =====
    for (int i = 0; i < MAX; i++) {
        for (tail = head[i].next; tail != NULL; tail = new) {
            new = tail->next;
            free(tail);
        }
    }

    return 0;
}
```

**运行效果：**
```
53 51 74 12 10 43 4 39 93 69
0 : 10                    ← 10%10=0
1 : 51                    ← 51%10=1
2 : 12                    ← 12%10=2
3 : 53 43 93              ← 53%10=3, 43%10=3, 93%10=3 (冲突→拉链)
4 : 74 4                  ← 74%10=4, 4%10=4
5 :
6 :
7 :
8 :
9 : 39 69                 ← 39%10=9, 69%10=9

input find key: 43
find: 43 (in bucket 3)
```

### 11.5 哈希表操作总结

| 操作 | 时间复杂度 | 说明 |
|------|-----------|------|
| 插入 | O(1) | 定位桶 + 头/尾插 |
| 查找 | O(1) 平均 | 定位桶 + 遍历链表 |
| 删除 | O(1) 平均 | 定位桶 + 链表删除 |

**拉链法要点：**
- 桶数组用头节点（有头单向），插入、删除操作统一
- 哈希函数质量决定性能 — 好的哈希函数使数据均匀分布
- 负载因子 = 数据个数 / 桶数量，过大时需扩容

## 12. 树 (Tree)

### 12.1 基本概念

**定义：** 树是一种**一对多**的数据结构，由节点和边组成。

**基本术语：**

| 术语 | 说明 |
|------|------|
| 根节点 | 没有父节点的节点（祖先节点） |
| 父节点 | 与子节点存在直接从属关系的上层节点 |
| 子节点 | 与父节点存在直接从属关系的下层节点 |
| 兄弟节点 | 拥有同一个父节点的两个子节点 |
| 叶子节点 | 没有子节点的节点 |
| 深度 | 从根节点到叶子节点经历的层次数 |

### 12.2 二叉树

**定义：** 每个节点最多拥有两个子节点的树。

**特殊二叉树：**

| 类型 | 定义 |
|------|------|
| 满二叉树 | 深度为 n，节点数满足 2ⁿ - 1 |
| 完全二叉树 | 上一层为满二叉树，第 n 层从右到左依次连续缺节点 |
| 平衡二叉树 | 每个节点的左右子树深度差绝对值 ≤ 1 |

### 12.3 二叉排序树 (BST)

**性质（核心规则）：**
- 所有左子树中的数据 < 根节点数据
- 所有右子树中的数据 ≥ 根节点数据
- 左子树中最大的数据 < 右子树中最小的数据

```
        50
       /  \
     30    70
    /  \   / \
   20  40 60  80
```

### 12.4 节点结构

```c
struct tree_t {
    int data;              // 数据域
    struct tree_t *left;   // 左子树（左孩子）
    struct tree_t *right;  // 右子树（右孩子）
};
```

### 12.5 插入操作

按照 BST 规则递归寻找合适位置：小的往左，大的往右（等于也往右）。

```c
void insert(struct tree_t **root, int data)
{
    struct tree_t *new = (struct tree_t *)malloc(sizeof(struct tree_t));
    new->data = data;
    new->left = new->right = NULL;

    if (*root == NULL) {
        *root = new;            // 空树：直接作为根
        return;
    }

    struct tree_t *tail = *root;
    while (1) {
        if (new->data < tail->data) {
            if (tail->left == NULL) {
                tail->left = new;    // 插入左子树
                break;
            }
            tail = tail->left;
        } else {                     // >= 走右边
            if (tail->right == NULL) {
                tail->right = new;   // 插入右子树
                break;
            }
            tail = tail->right;
        }
    }
}
```

**注意：** 使用二级指针 `**root`，因为空树时需要修改根指针本身。

### 12.6 树的遍历

#### 前序遍历（中左右）

```c
void fri(struct tree_t *root)
{
    if (root == NULL) return;
    printf("%d ", root->data);   // 中
    fri(root->left);             // 左
    fri(root->right);            // 右
}
```

#### 中序遍历（左中右）— BST 中序遍历结果有序

```c
void mid(struct tree_t *root)
{
    if (root == NULL) return;
    mid(root->left);             // 左
    printf("%d ", root->data);   // 中
    mid(root->right);            // 右
}
```

#### 后序遍历（左右中）

```c
void end(struct tree_t *root)
{
    if (root == NULL) return;
    end(root->left);             // 左
    end(root->right);            // 右
    printf("%d ", root->data);   // 中
}
```

#### 层次遍历（利用队列）

```c
void level(struct tree_t *root)
{
    struct tree_t *queue[MAX];
    int front = 0, end = 0;

    queue[end++] = root;           // 根入队

    while (front != end) {
        root = queue[front++];     // 出队并访问
        printf("%d ", root->data);

        if (root->left != NULL)    // 左孩子入队
            queue[end++] = root->left;
        if (root->right != NULL)   // 右孩子入队
            queue[end++] = root->right;
    }
}
```

#### 横向树形打印

```c
void show(struct tree_t *root)
{
    static int lev = 0;            // static：递归中保持状态

    if (root == NULL) return;

    lev++;
    show(root->right);             // 先打印右子树
    lev--;

    for (int i = 0; i < lev; i++)  // 缩进表示深度
        printf("    ");
    printf("%d\n", root->data);

    lev++;
    show(root->left);              // 再打印左子树
    lev--;
}
```

### 12.7 最值查找

```c
// 最大值：一直往右走
struct tree_t *max(struct tree_t *root)
{
    if (root == NULL) return NULL;
    while (root->right != NULL)
        root = root->right;
    return root;
}

// 最小值：一直往左走
struct tree_t *min(struct tree_t *root)
{
    if (root == NULL) return NULL;
    while (root->left != NULL)
        root = root->left;
    return root;
}
```

### 12.8 节点数与深度

```c
// 节点数（递归）
int count(struct tree_t *root)
{
    if (root == NULL) return 0;
    return 1 + count(root->left) + count(root->right);
}

// 树深度（递归）
int depth(struct tree_t *root)
{
    if (root == NULL) return 0;
    int ld = depth(root->left);
    int rd = depth(root->right);
    return 1 + (ld > rd ? ld : rd);
}
```

### 12.9 查找

利用 BST 性质：比当前节点大往右，比当前节点小往左。

```c
struct tree_t *find(struct tree_t *root, int key)
{
    while (root != NULL) {
        if (key > root->data)
            root = root->right;
        else if (key < root->data)
            root = root->left;
        else
            return root;           // 找到了
    }
    return NULL;
}
```

### 12.10 删除节点

BST 删除分三种情况：

```c
void del(struct tree_t **root, int key)
{
    struct tree_t *parent = NULL;
    struct tree_t *tail = *root;
    struct tree_t *new = NULL;

    // ===== 1. 查找目标节点及其父节点 =====
    while (tail != NULL) {
        if (key > tail->data) {
            parent = tail;
            tail = tail->right;
        } else if (key < tail->data) {
            parent = tail;
            tail = tail->left;
        } else {
            break;                 // 找到了
        }
    }
    if (tail == NULL) return;      // 没找到

    // ===== 2. 处理三种情况 =====
    if (tail->left == NULL && tail->right != NULL)      // 只有右孩子
        new = tail->right;
    else if (tail->left != NULL && tail->right == NULL) // 只有左孩子
        new = tail->left;
    else if (tail->left != NULL && tail->right != NULL) // 有两个孩子
    {
        // 策略 L→R：将左子树挂到右子树的最小节点下面
        min(tail->right)->left = tail->left;
        new = tail->right;
        // 或策略 R→L：将右子树挂到左子树的最大节点下面
        // max(tail->left)->right = tail->right;
        // new = tail->left;
    }
    // 叶子节点（左右都空）：new 保持 NULL

    // ===== 3. 将 new 挂到父节点 =====
    if (parent == NULL)
        *root = new;               // 删除的是根节点
    else if (key > parent->data)
        parent->right = new;
    else
        parent->left = new;

    free(tail);
}
```

**删除的三种情况图示：**
```
1. 叶子节点（无孩子）：直接删除
   删除 20:      50          50
                /            /
               20    →     NULL

2. 只有一个孩子：用孩子替换
   删除 30:      50          50
               /  \        /  \
              30  70  →   40  70
               \
               40

3. 有两个孩子（L→R 策略）：
   删除 50:
       50                   60
      /  \      →         /  \
     30   70             30   70
     /    /             /      \
    20   60            20       80  ← min(70)是60
           \            \
           80            (60->left = 30的左子树)
   L→R: min(右子树)->left = 左子树, 用右子树根替代
```

### 12.11 销毁

```c
void destroy(struct tree_t **root)
{
    if (*root == NULL) return;

    destroy(&((*root)->left));    // 递归销毁左子树
    destroy(&((*root)->right));   // 递归销毁右子树
    free(*root);
    *root = NULL;
}
```

**注意：** 必须用后序遍历销毁（先销毁孩子再销毁自己），否则内存泄漏。


## 13. 图 (Graph)

### 13.1 基本概念

**定义：** 图是一种**多对多**的数据结构，由顶点集和边集组成。

| 术语 | 说明 |
|------|------|
| 顶点 (Vertex) | 图中的数据节点 |
| 边 (Edge) | 两个顶点之间的连线 |
| 有向图 | 边有方向：A→B 不等于 B→A |
| 无向图 | 边无方向：A—B 等于 B—A |
| 权值 (Weight) | 边上的数值，表示距离/代价等 |
| 度 (Degree) | 无向图中与顶点相连的边数 |
| 入度/出度 | 有向图中指向/指出的边数 |
| 路径 | 从一个顶点到另一个顶点经过的边序列 |
| 连通图 | 任意两个顶点之间都有路径 |

**图示：**
```
无向图:          有向图:          带权图:
  A — B           A → B           A --5-- B
  |   |           ↓   ↓           |       |
  C — D           C ← D          2       3
                                   |       |
                                   C --4-- D
```

### 13.2 存储方式

#### 邻接矩阵

用二维数组 `arc[n][n]` 表示顶点间的连接关系：

```c
#define MAX_V 100

typedef struct graph_t {
    int vex[MAX_V];          // 顶点数据
    int arc[MAX_V][MAX_V];   // 邻接矩阵（0=无边, 1=有边, 或存权值）
    int num_vex;             // 顶点数
    int num_edge;            // 边数
} GRAPH;

// 无向图初始化示例
//    A(0) — B(1)
//     |      |
//    C(2) — D(3)
void init_graph(GRAPH *g)
{
    g->num_vex = 4;
    g->num_edge = 4;
    memset(g->arc, 0, sizeof(g->arc));

    g->arc[0][1] = g->arc[1][0] = 1;  // A-B
    g->arc[0][2] = g->arc[2][0] = 1;  // A-C
    g->arc[1][3] = g->arc[3][1] = 1;  // B-D
    g->arc[2][3] = g->arc[3][2] = 1;  // C-D
}
```

**优缺点：**
- 优点：O(1) 判断两点是否相连，实现简单
- 缺点：O(n²) 空间，稀疏图浪费严重

#### 邻接表

用链表数组表示每个顶点的邻接点：

```c
// 边节点
typedef struct edge_t {
    int adjvex;              // 邻接顶点下标
    int weight;              // 权值（可选）
    struct edge_t *next;
} EDGE;

// 顶点
typedef struct vex_t {
    int data;
    EDGE *first;             // 第一条邻接边
} VEX;

// 图
typedef struct graph2_t {
    VEX adjlist[MAX_V];
    int num_vex;
} GRAPH2;
```

**邻接表结构示意（无向图 A-B, A-C, B-D, C-D）：**
```
A(0): → B(1) → C(2)
B(1): → A(0) → D(3)
C(2): → A(0) → D(3)
D(3): → B(1) → C(2)
```

**优缺点：**
- 优点：O(n+e) 空间，适合稀疏图
- 缺点：判断两点是否相连需遍历链表

| 对比 | 邻接矩阵 | 邻接表 |
|------|----------|--------|
| 空间 | O(n²) | O(n+e) |
| 判断相邻 | O(1) | O(degree) |
| 遍历邻接点 | O(n) | O(degree) |
| 适用场景 | 稠密图 | 稀疏图 |

### 13.3 深度优先遍历 (DFS)

类似树的前序遍历，用**递归/栈**实现，一条路走到底再回溯。

```c
int visited[MAX_V] = {0};   // 访问标记数组

// 邻接矩阵 DFS
void dfs_matrix(GRAPH *g, int v)
{
    visited[v] = 1;
    printf("%d ", g->vex[v]);

    for (int i = 0; i < g->num_vex; i++) {
        if (g->arc[v][i] != 0 && !visited[i])
            dfs_matrix(g, i);            // 递归深入
    }
}

// 非递归版本（用栈）
void dfs_stack(GRAPH *g, int start)
{
    int stack[MAX_V], top = 0;
    int visited[MAX_V] = {0};

    stack[top++] = start;
    visited[start] = 1;

    while (top > 0) {
        int v = stack[--top];            // 出栈
        printf("%d ", g->vex[v]);

        for (int i = g->num_vex - 1; i >= 0; i--) {
            if (g->arc[v][i] != 0 && !visited[i]) {
                stack[top++] = i;        // 邻接点入栈
                visited[i] = 1;
            }
        }
    }
}
```

**DFS 遍历过程（从 A 出发）：**
```
    A — B
    |   |
    C — D

A → B → D → C
步骤: A入栈→出A, D/C入栈→出C→出D→出B
```

### 13.4 广度优先遍历 (BFS)

类似树的层次遍历，用**队列**实现，一层一层向外扩展。

```c
// 邻接矩阵 BFS
void bfs_matrix(GRAPH *g, int start)
{
    int queue[MAX_V], front = 0, end = 0;
    int visited[MAX_V] = {0};

    queue[end++] = start;                // 起点入队
    visited[start] = 1;

    while (front != end) {
        int v = queue[front++];          // 出队并访问
        printf("%d ", g->vex[v]);

        for (int i = 0; i < g->num_vex; i++) {
            if (g->arc[v][i] != 0 && !visited[i]) {
                queue[end++] = i;        // 邻接点入队
                visited[i] = 1;
            }
        }
    }
}
```

**BFS 遍历过程（从 A 出发）：**
```
    A — B
    |   |
    C — D

A → B → C → D
步骤: A入队→出A, B/C入队→出B, D入队→出C→出D
```

**DFS vs BFS：**
| 对比 | DFS | BFS |
|------|-----|-----|
| 数据结构 | 栈（递归） | 队列 |
| 遍历方式 | 纵深优先 | 层次优先 |
| 适用 | 路径查找、连通性 | 最短路径（无权）、层序遍历 |

### 13.5 最短路径 — Dijkstra 算法

求**单源最短路径**：从一个顶点出发到其他所有顶点的最短距离（权值非负）。

```c
#define INF 0x3f3f3f3f    // 无穷大

void dijkstra(GRAPH *g, int start)
{
    int dist[MAX_V];           // dist[i] = start 到 i 的最短距离
    int visited[MAX_V] = {0};  // 已确定最短路径的顶点
    int prev[MAX_V];           // 前驱节点（用于回溯路径）

    // 初始化
    for (int i = 0; i < g->num_vex; i++) {
        dist[i] = (g->arc[start][i] != 0) ? g->arc[start][i] : INF;
        prev[i] = start;
    }
    dist[start] = 0;
    visited[start] = 1;

    // 每次确定一个顶点的最短路径
    for (int count = 1; count < g->num_vex; count++) {
        // 找到未访问中距离最小的顶点
        int min = INF, u = -1;
        for (int i = 0; i < g->num_vex; i++) {
            if (!visited[i] && dist[i] < min) {
                min = dist[i];
                u = i;
            }
        }
        if (u == -1) break;    // 剩下的不可达

        visited[u] = 1;        // u 的最短路径已确定

        // 以 u 为跳板更新其他顶点的距离
        for (int v = 0; v < g->num_vex; v++) {
            if (!visited[v] && g->arc[u][v] != 0
                && dist[u] + g->arc[u][v] < dist[v]) {
                dist[v] = dist[u] + g->arc[u][v];
                prev[v] = u;
            }
        }
    }

    // 输出结果
    for (int i = 0; i < g->num_vex; i++)
        printf("0→%d: %d\n", i, dist[i]);
}
```

**算法过程：**
```
      A --5-- B
      |      /|
      2     3 1
      |  /    |
      C --4-- D

从 A(0) 出发:
  初始: dist=[0, 5, 2, INF]
  确定 C(2): 以 C 更新 → dist=[0, 5, 2, 6]    (A→C→D=6)
  确定 B(5): 以 B 更新 → dist=[0, 5, 2, 6]    (A→C→B=5,不更新)
  确定 D(6): 完成
  结果: A→A:0, A→B:5, A→C:2, A→D:6
```

### 13.6 最小生成树 — Prim 算法

在带权连通图中找一棵包含所有顶点且总权值最小的生成树。

```c
void prim(GRAPH *g, int start)
{
    int lowcost[MAX_V];        // lowcost[i] = 到 i 的最小代价
    int closest[MAX_V];        // closest[i] = 最近的已选点
    int visited[MAX_V] = {0};

    // 初始化
    for (int i = 0; i < g->num_vex; i++) {
        lowcost[i] = (g->arc[start][i] != 0) ? g->arc[start][i] : INF;
        closest[i] = start;
    }
    visited[start] = 1;

    // 每次选一条最小边加入生成树
    for (int count = 1; count < g->num_vex; count++) {
        int min = INF, u = -1;
        for (int i = 0; i < g->num_vex; i++) {
            if (!visited[i] && lowcost[i] < min) {
                min = lowcost[i];
                u = i;
            }
        }
        if (u == -1) break;

        visited[u] = 1;
        printf("边: %d-%d 权值:%d\n", closest[u], u, lowcost[u]);

        // 以 u 为基准更新 lowcost
        for (int v = 0; v < g->num_vex; v++) {
            if (!visited[v] && g->arc[u][v] != 0
                && g->arc[u][v] < lowcost[v]) {
                lowcost[v] = g->arc[u][v];
                closest[v] = u;
            }
        }
    }
}
```

**Prim 过程：**
```
      A --5-- B
      |      /|
      2     3 1
      |  /    |
      C --4-- D

初始: 从 A 开始, lowcost=[0, 5, 2, INF]
第1步: 选 C(边 A-C=2) → lowcost=[0, 3, 2, 4]
第2步: 选 B(边 C-B=3) → lowcost=[0, 3, 2, 1]
第3步: 选 D(边 B-D=1) → 完成
MST 边: A-C(2), C-B(3), B-D(1)  总权值=6
```

### 13.7 完整演示

```c
#include <stdio.h>
#include <string.h>

#define MAX_V 100
#define INF 0x3f3f3f3f

typedef struct {
    int vex[MAX_V];
    int arc[MAX_V][MAX_V];
    int num_vex, num_edge;
} GRAPH;

void bfs(GRAPH *g, int start)
{
    int queue[MAX_V], front = 0, end = 0;
    int visited[MAX_V] = {0};

    queue[end++] = start;
    visited[start] = 1;
    printf("BFS: ");

    while (front != end) {
        int v = queue[front++];
        printf("%d ", v);
        for (int i = 0; i < g->num_vex; i++) {
            if (g->arc[v][i] != 0 && !visited[i]) {
                queue[end++] = i;
                visited[i] = 1;
            }
        }
    }
    printf("\n");
}

int main(void)
{
    GRAPH g;
    g.num_vex = 5;

    // 邻接矩阵（带权无向图）
    int matrix[5][5] = {
        {0, 5, 2, 0, 0},   // A: →B(5), →C(2)
        {5, 0, 3, 1, 0},   // B: →A(5), →C(3), →D(1)
        {2, 3, 0, 4, 0},   // C: →A(2), →B(3), →D(4)
        {0, 1, 4, 0, 6},   // D: →B(1), →C(4), →E(6)
        {0, 0, 0, 6, 0},   // E: →D(6)
    };
    memcpy(g.arc, matrix, sizeof(matrix));

    bfs(&g, 0);                  // BFS: 0 1 2 3 4

    return 0;
}
```

### 13.9 数据结构学习总结

回顾整个数据结构的学习路线：

```
1. 顺序表 (DARR)        — 连续存储，随机访问
2. 单向链表              — 不连续存储，插入删除 O(1)
3. 双向链表              — 支持前后遍历
4. 链表框架 (LLIST)       — 泛型封装
5. 栈 (Stack)            — FILO，退格/表达式求值
6. 队列 (Queue)          — FIFO，循环队列，BFS
7. 排序算法              — 冒泡/选择/插入/快排
8. 查找算法              — 顺序/二分/哈希
9. 哈希表               — O(1) 查找，空间换时间
10. 树 (BST)             — 一对多，有序存储
11. 图                   — 多对多，网络关系

每种结构的核心操作: 增、删、查、改、遍历、销毁
```

**数据结构选择的经验法则：**

| 需求 | 推荐结构 |
|------|----------|
| 频繁随机访问 | 顺序表 |
| 频繁插入/删除 | 链表 |
| 需要 O(1) 查找 | 哈希表 |
| 需要排序/范围查询 | 二叉排序树 |
| 先进后出 | 栈 |
| 先进先出 | 队列 |
| 多对多关系 | 图 |
